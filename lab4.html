# docker/promotion/promotion_server.py
import os
import ssl
import threading
import urllib3
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from kubernetes import client, config

click_count = 0
THRESHOLD = 3
lock = threading.Lock()

NAMESPACE = "dev"
ROUTE_NAME = "ab-test-route"


def initialize_kube_client():
    try:
        # Load the in-cluster configuration
        config.load_incluster_config()
        configuration = client.Configuration.get_default_copy()

        ca_cert = '/var/run/secrets/kubernetes.io/serviceaccount/ca.crt'

        # 1. Create a modern SSL Context for Python 3.14
        context = ssl.create_default_context(cafile=ca_cert)

        # 2. THE PYTHON 3.14 FIX: Disable 'STRICT' mode
        # This allows MicroK8s certificates (which lack KeyUsage extensions)
        # while still verifying the certificate chain (SSL=True).
        context.verify_flags &= ~ssl.VERIFY_X509_STRICT

        # 3. Lower security level to allow legacy certificate formats
        context.set_ciphers('DEFAULT@SECLEVEL=1')

        # 4. Inject the custom context into the API client's connection pool
        api_client = client.ApiClient(configuration)
        api_client.rest_client.pool_manager.connection_pool_kw['ssl_context'] = context

        print("Kubernetes API client initialized with SSL=True (Python 3.14 Fix Applied).")
        # CustomObjectsApi is needed to patch Gateway API resources (HTTPRoute)
        return client.CustomObjectsApi(api_client)

    except Exception as e:
        print(f"Initial setup failed: {e}")
        # Crash early if we can't get a secure connection
        raise


api_instance = initialize_kube_client()


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        global click_count
        message = ""

        if self.path == "/kill":
            with lock:
                click_count += 1
                count = click_count

            message = f"Click registered ({count}/{THRESHOLD})"
            print(message, flush=True)

            if count >= THRESHOLD:
                message += " -> Promotion triggered! v2 now 100%"
                print(message, flush=True)

                if api_instance:
                    try:
                        # JSON Patch: v1 (index 0) weight -> 0, v2 (index 1) weight -> 100
                        # Indexes match the order of backendRefs in k8s/canary.yaml
                        body = [
                            {"op": "replace",
                             "path": "/spec/rules/0/backendRefs/0/weight",
                             "value": 0},
                            {"op": "replace",
                             "path": "/spec/rules/0/backendRefs/1/weight",
                             "value": 100},
                        ]

                        print(f"Patching HTTPRoute {ROUTE_NAME} in {NAMESPACE}...", flush=True)

                        api_instance.patch_namespaced_custom_object(
                            group="gateway.networking.k8s.io",
                            version="v1",
                            namespace=NAMESPACE,
                            plural="httproutes",
                            name=ROUTE_NAME,
                            body=body,
                            _content_type="application/json-patch+json",
                            _request_timeout=5
                        )

                        message += " HTTPRoute updated successfully."
                        print("HTTPRoute patch applied.", flush=True)

                    except Exception as e:
                        message += f" K8s API Error: {e}"
                        print(f"Exception during patch: {e}", flush=True)
                else:
                    message += " API client unavailable."

            self.send_response(200)
            self.send_header("Content-type", "text/plain")
            self.end_headers()
            self.wfile.write(message.encode())

        elif self.path == "/health":
            self.send_response(200)
            self.end_headers()
            self.wfile.write(b"OK")
        else:
            self.send_response(404)
            self.end_headers()


if __name__ == "__main__":
    print("Starting Promotion Server...", flush=True)
    server = ThreadingHTTPServer(("0.0.0.0", 5000), Handler)
    print("Promotion server running on port 5000", flush=True)
    server.serve_forever()
# docker/v1/app.py
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import os
import requests  # Importing the requests module to make HTTP calls

# URL of the promotion server (adjust as needed)
PROMOTION_SERVER_URL = "http://promotion-server:5000/kill"


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/":
            self.send_response(200)
            self.send_header("Content-type", "text/html")
            self.end_headers()
            with open("index.html", "rb") as f:
                self.wfile.write(f.read())

        elif self.path == "/kill":
            # Log when /kill is hit
            print("Received /kill request, calling promotion server...", flush=True)

            # Send a valid HTTP response first, then crash
            self.send_response(200)
            self.send_header("Content-type", "text/plain")
            self.end_headers()

            try:
                response = requests.get(PROMOTION_SERVER_URL, timeout=3)
                if response.status_code == 200:
                    print("Promotion server count updated!", flush=True)
                    self.wfile.write(b"Promotion server count updated! ")
                else:
                    print(f"Failed to update promotion server count. Status code: {response.status_code}", flush=True)
                    self.wfile.write(b"Failed to update promotion server count. ")
            except Exception as e:
                print(f"Error calling promotion server: {e}", flush=True)
                self.wfile.write(b"Error calling promotion server. ")

            # Simulate crash
            self.wfile.write(b"Crashing pod...")
            self.wfile.flush()
            os._exit(1)

        elif self.path == "/health":
            self.send_response(200)
            self.end_headers()
            self.wfile.write(b"OK")

        else:
            self.send_response(404)
            self.end_headers()


if __name__ == "__main__":
    server = ThreadingHTTPServer(("0.0.0.0", 8080), Handler)
    print("Self-healing server v1 running on port 8080", flush=True)
    server.serve_forever()

6. Test the lab

Open http://127.0.0.1:8081 on your computer (bridged network: http://<VM IP>). Refresh many times: about 70% v1, 30% v2.

1. Click "Trigger Bug" on v1 -> the pod crashes and Kubernetes restarts it
   (watch: sudo microk8s kubectl get pods -n dev).
2. Each click also increments the promotion server counter.
3. After 3 clicks the HTTPRoute is patched -> 100% of traffic goes to v2.
4. v2 shows the "Bug Free!!" button.

Run it again: reapply the route and reset the counter
   sudo microk8s kubectl apply -f k8s/canary.yaml
   sudo microk8s kubectl delete pods --all -n dev
