DevSecOps Observability Lab

Architecture

  • One Prometheus scrapes infrastructure + app metrics

  • Alloy collects pipeline + app logs

  • Loki stores logs from both

  • Grafana dashboards visualize all metrics

  • Chaos scenarios to trigger alerts

3️⃣ Jenkinsfile 

pipeline {
    agent any // This ensures a node/executor is allocated
    stages {
        stage('Build') {
            steps {
                script {
                    try {
                        // All 'sh' steps must happen inside an agent/node context. This will always fail without a makefile
                        sh 'make build'
                    } catch (err) {
                        // The 'node' context is still active here, so this 'sh' works
                        def lokiUrl = "http://loki.observability.svc.cluster.local:3100/loki/api/v1/push"
                        def timestamp = System.currentTimeMillis() * 1000000
                        
                        sh """
                        curl -X POST -H "Content-Type: application/json" \
                        -d '{"streams": [{"stream": {"job": "jenkins-pipeline"}, "values": [["${timestamp}", "Build Failed: ${err}"]]}]}' \
                        ${lokiUrl}
                        """
                        error("Build failed: ${err}")
                    }
                }
            }
        }
    }
}

demo-app/demo-app-configmap.yaml

apiVersion: v1
kind: ConfigMap
metadata:
  name: demo-app-html
  namespace: observability
data:
  index.html: |
    <!DOCTYPE html>
    <html lang="en">
    <head>
        <meta charset="UTF-8">
        <meta name="viewport" content="width=device-width, initial-scale=1.0">
        <title>Demo App — Observability Lab</title>
        <style>
          body { font-family: monospace; padding: 2rem; background: #111; color: #0f0; }
          #log { margin-top: 1rem; font-size: 0.85rem; max-height: 300px; overflow-y: auto; }
          .ok  { color: #0f0; }
          .err { color: #f44; }
        </style>
    </head>
    <body>
        <h2>Demo App — Observability Lab</h2>
        <p>Generating traffic every <strong>5s</strong>. Watch Loki + Prometheus.</p>
        <div id="log"></div>
        <script>
          const log = document.getElementById('log');
          let tick = 0;

          function addEntry(msg, ok) {
            const line = document.createElement('div');
            line.className = ok ? 'ok' : 'err';
            line.textContent = `[${new Date().toISOString()}] ${msg}`;
            log.prepend(line);
            while (log.children.length > 50) log.removeChild(log.lastChild);
          }

          async function poll() {
            tick++;
            try {
              const r = await fetch('/');
              addEntry(`GET / → ${r.status}`, r.ok);
            } catch (e) {
              addEntry(`GET / → network error: ${e.message}`, false);
            }
            if (tick % 3 === 0) {
              try {
                const r = await fetch('/missing');
                addEntry(`GET /missing → ${r.status}`, r.ok);
              } catch (e) {
                addEntry(`GET /missing → network error: ${e.message}`, false);
              }
            }
          }

          poll();
          setInterval(poll, 5000);
        </script>
    </body>
    </html>

  default.conf: |
    server {
        listen 80;
        server_name _;

        location / {
            root   /usr/share/nginx/html;
            index  index.html;
        }

        location /nginx_status {
            stub_status on;
            allow 127.0.0.1;
            allow 10.0.0.0/8;
            deny all;
        }

        location = /favicon.ico {
            return 204;
            access_log off;
            log_not_found off;
        }
    }

prometheus/prometheus-alert-rules.yaml

apiVersion: v1
kind: ConfigMap
metadata:
  name: prometheus-alert-rules
  namespace: observability
data:
  alert-rules.yaml: |
    groups:
      - name: lab-infrastructure
        rules:

          # Any scrape target has gone down
          - alert: TargetDown
            expr: up == 0
            for: 1m
            labels:
              severity: critical
            annotations:
              summary: "Scrape target {{ $labels.job }} is down"
              description: "Prometheus cannot reach {{ $labels.job }} at {{ $labels.instance }}"

          # Jenkins disappeared entirely from scrape targets
          - alert: JenkinsMissing
            expr: absent(up{job="jenkins"})
            for: 2m
            labels:
              severity: critical
            annotations:
              summary: "Jenkins is not present in scrape targets"
              description: "No Jenkins instance is reachable — pipeline jobs cannot run"

          # Loki disappeared — logs are being lost
          - alert: LokiMissing
            expr: absent(up{job="loki"})
            for: 2m
            labels:
              severity: warning
            annotations:
              summary: "Loki is not reachable"
              description: "Log ingestion may be failing — check Alloy pipeline"

          # Alloy disappeared — logs are not being shipped
          - alert: AlloyMissing
            expr: absent(up{job="alloy"})
            for: 2m
            labels:
              severity: warning
            annotations:
              summary: "Alloy log shipper is not reachable"
              description: "Container logs may not be reaching Loki"

          # Demo app disappeared
          - alert: DemoAppDown
            expr: absent(up{job="demo-app"})
            for: 1m
            labels:
              severity: warning
            annotations:
              summary: "Demo app is not reachable"
              description: "The demo-app nginx service has disappeared from scrape targets"

          # Prometheus rule evaluation slow
          - alert: PrometheusSlowRuleEvaluation
            expr: prometheus_rule_evaluation_duration_seconds{quantile="0.9"} > 5
            for: 5m
            labels:
              severity: warning
            annotations:
              summary: "Prometheus rule evaluation is slow"
              description: "90th percentile rule evaluation is over 5s — check rule complexity"

logging/alloy-config.yaml


apiVersion: v1
kind: ConfigMap
metadata:
  name: alloy-config
  namespace: observability
data:
  config.alloy: |
    // ── LOG COLLECTION (Loki) ────────────────────────────────────────────────
    // Tails the actual log files on the node disk
    local.file_match "containers" {
      path_targets = [{
        __path__ = "/var/log/containers/*.log",
      }]
      sync_period = "10s"
    }

    loki.source.file "container_logs" {
      targets    = local.file_match.containers.targets
      forward_to = [loki.process.add_labels.receiver]
    }

    loki.process "add_labels" {
      stage.regex {
        expression = "/var/log/containers/(?P<pod>[^_]+)_(?P<namespace>[^_]+)_(?P<container>.+)-[0-9a-f]{64}[.]log"
        source     = "filename"
      }
      stage.labels {
        values = {
          pod       = "pod",
          namespace = "namespace",
          container = "container",
        }
      }
      forward_to = [loki.write.loki_endpoint.receiver]
    }

    loki.write "loki_endpoint" {
      endpoint {
        url = "http://loki:3100/loki/api/v1/push"
      }
    }

    // ── METRICS COLLECTION (Prometheus via cAdvisor) ─────────────────────────
    // This talks to the Kubelet to get CPU/RAM without needing extra exporters
    
    discovery.kubernetes "nodes" {
      role = "node"
    }

    prometheus.scrape "cadvisor" {
      targets = discovery.kubernetes.nodes.targets
      
      metrics_path = "/metrics/cadvisor"
      scheme       = "https"

      authorization {
        type             = "Bearer"
        credentials_file = "/var/run/secrets/kubernetes.io/serviceaccount/token"
      }

      tls_config {
        ca_file              = "/var/run/secrets/kubernetes.io/serviceaccount/ca.crt"
        insecure_skip_verify = true
      }

      forward_to = [prometheus.remote_write.prom_endpoint.receiver]
    }

    // ── SINGLE PUSH ENDPOINT ─────────────────────────────────────────────────
    // Consolidated to avoid the "Duplicate Component" error

    prometheus.remote_write "prom_endpoint" {
      endpoint {
        url = "http://prometheus:9090/api/v1/write"
      }
    }

  • In Manage Jenkins → Configure System → Clouds → Add a new cloud → Kubernetes:

      • Working Directory: /home/jenkins/agent
      • Leave Command and Arguments empty
